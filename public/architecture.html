Step 0 · Prepare the application before the workflow

Use your own Anthropic API key and an existing agent ID and environment ID.

Run npm run setup once after setting your API key. It creates an environment and base agent and writes their IDs into your local .env. Reuse those IDs on later runs.

// One-time setup: an Environment and a base Agent that DOES have capabilities
// (the built-in agent toolset), so the session override on slide 07 has
// something to clear. Agents are persistent - run once, reuse the IDs.
import Anthropic from "@anthropic-ai/sdk";
import { existsSync, readFileSync, writeFileSync } from "node:fs";

const client = new Anthropic();

const environment = await client.beta.environments.create({
  name: "lesson-c03-env",
  config: { type: "cloud", networking: { type: "limited" } },
});

const agent = await client.beta.agents.create({
  name: "Lesson C03 Base Agent",
  model: process.env.CLAUDE_MODEL ?? "claude-opus-5-5",
  system: "You are a thorough research assistant. Explain your reasoning in detail.",
  tools: [{ type: "agent_toolset_20260401" }],
});

const ids = {
  agentId: agent.id,
  agentVersion: agent.version,
  environmentId: environment.id,
};
writeFileSync(".managed-ids.json", JSON.stringify(ids, null, 2));

// The slides read the IDs from process.env, so write them into .env too.
const env = existsSync(".env") ? readFileSync(".env", "utf8") : "";
const upsert = (text: string, key: string, value: string) => {
  const line = `${key}=${value}`;
  const re = new RegExp(`^#?\s*${key}=.*$`, "m");
  return re.test(text) ? text.replace(re, line) : `${text.trimEnd()}\n${line}\n`;
};
let next = upsert(env, "CLAUDE_MANAGED_AGENT_ID", agent.id);
next = upsert(next, "CLAUDE_MANAGED_ENVIRONMENT_ID", environment.id);
writeFileSync(".env", next);

console.log("Base agent:", {
  id: agent.id,
  version: agent.version,
  model: agent.model,
  tools: agent.tools.length,
});
console.log("Saved .managed-ids.json and updated .env:", ids);

const $ = (selector) => document.querySelector(selector);

  .then((r) => r.json())
  .then((c) => {
    button.disabled = !c.configured;
    status.textContent = c.configured
      ? "Configuration ready. Your API key stays on the server."
      : "Configure your API key and resource IDs in .env.";
  })
  .catch(() => {
    button.disabled = true;
    status.textContent = "Could not check configuration.";

button.addEventListener("click", async () => {
  const prompt = $("#prompt").value.trim();
  if (!prompt) {
    status.textContent = "Enter a prompt.";
    return;
  }
  button.disabled = true;
  status.textContent = "Creating session…";
  $("#events").replaceChildren();
  $("#history").textContent = "Waiting for the session to finish.";
  $("#answer").hidden = true;
  $("#answer").textContent = "";
  $("#stored").textContent = "Loading…";
  let sawStop = false,
    failed = false;
  const handle = ({ type, value }) => {
    if (type === "session") {
      $("#empty").hidden = true;
      $("#result").hidden = false;
      $("#session").textContent = value.id;
      $("#agent").textContent = `${value.agentId} · version ${value.version}`;
      $("#effective").textContent =
        `Tools ${value.tools} · MCP ${value.mcp} · Skills ${value.skills}`;
      status.textContent = "Listening for live events…";
    }
    if (type === "stored")
      $("#stored").textContent =
        `Tools ${value.tools} · MCP ${value.mcp} · Skills ${value.skills} · version ${value.version}`;
    if (type === "event") {
      const li = document.createElement("li");
      li.textContent = `${value.type} · ${value.id}`;
      $("#events").append(li);
    }
    if (type === "answer") {
      $("#answer").hidden = false;
      $("#answer").textContent += value;
    }
    if (type === "history") {
      $("#history").textContent =
        `${value.length} events recorded: ${value.map((e) => e.type).join(", ")}`;
    }
    if (type === "stop") {
      sawStop = true;
      status.textContent = `Session stopped: ${value}.`;
    }
    if (type === "error") {
      failed = true;
      status.textContent = value;
    }
  };
  try {
    const response = await fetch("/api/run", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ prompt }),
    });
    if (!response.ok) {
      const data = await response.json();
      throw new Error(data.error || "Request failed");
    }
    const reader = response.body.getReader(),
      decoder = new TextDecoder();
    let buffer = "";
    while (true) {
      const { done, value } = await reader.read();
      buffer += decoder.decode(value || new Uint8Array(), { stream: !done });
      const lines = buffer.split("\n");
      buffer = lines.pop();
      for (const line of lines) if (line) handle(JSON.parse(line));
      if (done) break;
    }
    if (buffer.trim()) handle(JSON.parse(buffer));
    if (!sawStop && !failed)
      status.textContent =
        "Stream ended before a stop event; inspect the recorded session.";
  } catch (error) {
    status.textContent = error.message;
  } finally {
    button.disabled = false;
  }
});

import { serveStudentAsset } from "./student-assets.js";
import Anthropic from "@anthropic-ai/sdk";
import { createServer } from "node:http";
import { readFile } from "node:fs/promises";
import { writeFileSync } from "node:fs";

const port = Number(process.env.PORT || 3003);
const client = new Anthropic({ apiKey: process.env.ANTHROPIC_API_KEY || "not-configured" });
const root = new URL("../", import.meta.url);
const configured = Boolean(process.env.ANTHROPIC_API_KEY && process.env.CLAUDE_MANAGED_AGENT_ID && process.env.CLAUDE_MANAGED_ENVIRONMENT_ID);

createServer(async (req, res) => {
  const pathname = new URL(req.url || "/", "http://localhost").pathname;
  const assets: Record<string, [string, string]> = { "/": ["index.html", "text/html; charset=utf-8"], "/style.css": ["style.css", "text/css; charset=utf-8"], "/app.js": ["app.js", "text/javascript; charset=utf-8"], "/astronaut.png": ["astronaut.png", "image/png"] };
  try {
    if (await serveStudentAsset(req, res)) return;
    if (req.method === "GET" && assets[pathname]) {
      const [file, type] = assets[pathname]; const content = await readFile(new URL(`public/${file}`, root));
      res.writeHead(200, { "Content-Type": type, "Cache-Control": "no-store" }); res.end(content); return;
    }
    if (req.method === "GET" && pathname === "/api/config") {
      res.writeHead(200, { "Content-Type": "application/json" }); res.end(JSON.stringify({ configured })); return;
    }
    if (req.method === "POST" && pathname === "/api/run") {
      const origin = req.headers.origin;
      if (origin && origin !== `http://127.0.0.1:${port}` && origin !== `http://localhost:${port}`) { res.writeHead(403).end(); return; }
      let body = "";
      for await (const chunk of req) { body += chunk; if (body.length > 10000) throw new Error("Request too large"); }

        send("session", { id: session.id, agentId: session.agent.id, version: session.agent.version, tools: session.agent.tools.length, mcp: session.agent.mcp_servers.length, skills: session.agent.skills.length });
        const stored = await client.beta.agents.retrieve(session.agent.id);
        send("stored", { version: stored.version, tools: stored.tools.length, mcp: stored.mcp_servers.length, skills: stored.skills.length });
        const live = await client.beta.sessions.events.stream(session.id);
        res.on("close", () => live.controller.abort());
        await client.beta.sessions.events.send(session.id, { events: [{ type: "user.message", content: [{ type: "text", text: data.prompt.trim() }] }] });
        for await (const event of live) {
          send("event", { id: "id" in event ? event.id : "stream update", type: event.type });
          if (event.type === "agent.message") for (const block of event.content) if (block.type === "text") send("answer", block.text);
          if (event.type === "session.status_idle" && event.stop_reason.type !== "requires_action") { send("stop", event.stop_reason.type); break; }
          if (event.type === "session.status_terminated") { send("stop", "terminated"); break; }
          if (event.type === "session.error") { send("error", "The session reported an error. Inspect its history for details."); break; }
        }
        const history: { id: string; type: string }[] = [];
        for await (const event of client.beta.sessions.events.list(session.id)) history.push({ id: event.id, type: event.type });
        send("history", history);
      } catch (error) { send("error", error instanceof Error ? error.message : "Run failed"); }
      res.end(); return;
    }
    res.writeHead(404).end("Not found");
